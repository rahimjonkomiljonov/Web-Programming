ProfileCard with a default prop

ProfileCard.jsx

Goal: build a card with name, role, email and an optional avatarUrl that has a default value.

// Study guide, exercise 1: a ProfileCard with name, role, email and an optional avatarUrl.
// avatarUrl has a default value, so the card still works when the parent does not pass one.
// (The guide's default pointed at via.placeholder.com, which no longer exists, so the default
// here is a small grey person drawn as an inline SVG.)
const DEFAULT_AVATAR =
  'data:image/svg+xml,' +
  encodeURIComponent(
    '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 80 80">' +
      '<rect width="80" height="80" fill="#ddd3bf"/>' +
      '<circle cx="40" cy="31" r="14" fill="#5c564a"/>' +
      '<path d="M12 80c2-18 14-28 28-28s26 10 28 28z" fill="#5c564a"/></svg>',
  );

function ProfileCard({ name, role, email, avatarUrl = DEFAULT_AVATAR }) {
  return (
    <div className="card profile-card">
      <img className="avatar" src={avatarUrl} alt={name} width="80" height="80" />
      <h3>{name}</h3>
      <p>{role}</p>
      <a href={`mailto:${email}`}>{email}</a>
    </div>
  );
}

export default ProfileCard;

E4

List of notifications

NotificationList.jsx

Goal: render an array of notifications with .map(), a stable key and an icon chosen by conditional rendering.

// Study guide, exercise 2: render a list of notifications with conditional icons.
//   success -> check mark, warning -> warning sign, anything else -> info.
// Each item gets a stable key from its id (never the array index).
const notifications = [
  { id: 1, type: 'success', message: 'Profile updated successfully' },
  { id: 2, type: 'warning', message: 'Password expires in 3 days' },
  { id: 3, type: 'info', message: 'New lecture materials available' },
];

function NotificationItem({ type, message }) {
  const icon = type === 'success' ? '✅' : type === 'warning' ? '⚠️' : 'ℹ️';
  return (
    <li className={`notification notification-${type}`}>
      <span aria-hidden="true">{icon}</span>
      <span className="sr-only">{type}: </span>
      {message}
    </li>
  );
}

function NotificationList() {
  return (
    <ul className="notifications">
      {notifications.map((n) => (
        <NotificationItem key={n.id} type={n.type} message={n.message} />
      ))}
    </ul>
  );
}

export default NotificationList;

Result (running)

Result: three notifications with a check mark, a warning sign and an info icon. The key is each item's id, not its position. Open it on the practice page.

E6

ProductCard code challenge

ProductCard.jsx

Goal: build a ProductCard with name, price, rating and inStock. Bonus 1: show the rating as “4.5 / 5 stars”. Bonus 2: show an “Out of Stock” label when inStock is false.

// In-class code challenge: build a ProductCard.
// Props: name (string), price (number, KRW), rating (number, 0 to 5), inStock (boolean).
// Bonus 1: show the rating as "4.5 / 5 stars".
// Bonus 2: show an "Out of Stock" label when inStock is false.
function Stars({ rating }) {
  const full = Math.floor(rating);
  const half = rating - full >= 0.5 ? 1 : 0;
  const empty = 5 - full - half;
  // the stars are only decoration: the words "4.5 / 5 stars" carry the meaning
  return (
    <span className="stars" aria-hidden="true">
      {'★'.repeat(full)}
      {half ? '½' : ''}
      {'☆'.repeat(empty)}
    </span>
  );
}

function ProductCard({ name, price, rating, inStock }) {
  return (
    <div className={inStock ? 'card product-card' : 'card product-card sold-out'}>
      <h3>{name}</h3>
      <p className="price">{price.toLocaleString('ko-KR')}원</p>
      <Stars rating={rating} />
      <p className="rating">{rating} / 5 stars</p>
      <p className={inStock ? 'stock in-stock' : 'stock out-of-stock'}>
        {inStock ? 'In Stock' : 'Out of Stock'}
      </p>
    </div>
  );
}

export default ProductCard;

// App.jsx
<ProductCard name="Wireless Earbuds" price={89000} rating={4.5} inStock={true} />
<ProductCard name="Mechanical Keyboard" price={129000} rating={4} inStock={true} />
<ProductCard name="USB-C Hub" price={45000} rating={3.5} inStock={false} />

Result (running)

Result: three cards. Prices are formatted in won (89,000원), the ratings read “4.5 / 5 stars”, and the USB-C Hub shows “Out of Stock” and is greyed out. Both bonuses are done. Open it on the practice page.

E8

Conditional rendering

ConditionalDemo.jsx

Goal: show different content depending on state: return null, use && and use a ternary.

import { useState } from 'react';

// Lecture example: conditional rendering three ways.
//   Alert      -> returns null to render nothing when there is no message
//   Dashboard  -> && shows a part only when its condition is true
//   the button -> a ternary picks one of two labels
function Alert({ type, message }) {
  if (!message) return null; // render nothing
  return (
    <div className={`alert alert-${type}`} role="status">
      {type === 'error' ? 'ERROR:' : 'OK:'} {message}
    </div>
  );
}

function Dashboard({ user }) {
  return (
    <div>
      {user && <p>Welcome, {user.name}!</p>}
      {!user && <p>Please log in.</p>}
    </div>
  );
}

// useState is only used here to flip the login state so both branches can be seen
function ConditionalDemo() {
  const [user, setUser] = useState(null);
  return (
    <div>
      <Alert type="error" message={user ? '' : 'You are not logged in.'} />
      <Alert type="success" message={user ? 'Logged in successfully.' : ''} />
      <div className="card">
        <Dashboard user={user} />
        <button type="button" onClick={() => setUser(user ? null : { name: 'Rahimjon' })}>
          {user ? 'Log out' : 'Log in'}
        </button>
      </div>
    </div>
  );
}

export default ConditionalDemo;

Result (running)

Result: with nobody logged in the page shows the error and “Please log in.”; the button logs in and the page shows the success message and “Welcome, Rahimjon!”. This is the only place I needed useState, to flip between the two states. Open it on the practice page.
